# 解答例

各モジュールのコードをそのままコピーして、VBE の標準モジュールに貼り付けてください。

モジュールは5つに分けて作るのがおすすめです（`M01_基礎` 〜 `M05_総合`）。

> 解答は「一つの正解」ではありません。動けば正解です。ここでは**実務で壊れにくい書き方**を優先しています。

Sub Q4_金額を計算()
    Dim ws As Worksheet
    Dim price As Long          ' 単価（整数）
    Dim qty As Long            ' 数量（整数）
    Dim amount As Long         ' 金額
    Dim taxIncluded As Double  ' 税込（小数が出るので Double）
    Const TAX_RATE As Double = 0.1

    Set ws = ThisWorkbook.Worksheets("売上明細")
    price = ws.Range("I2").Value
    qty = ws.Range("J2").Value

    amount = price * qty
    taxIncluded = amount * (1 + TAX_RATE)

    MsgBox "単価: " & Format(price, "#,##0") & " 円" & vbLf & _
           "数量: " & qty & vbLf & _
           "金額: " & Format(amount, "#,##0") & " 円" & vbLf & _
           "税込: " & Format(taxIncluded, "#,##0") & " 円"
End Sub

Sub Q5_発注判定()
    Dim ws As Worksheet
    Dim lastRow As Long, i As Long

    Set ws = ThisWorkbook.Worksheets("在庫")
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

        If ws.Cells(i, "D").Value = 0 Then

        End If
    Next i

Sub Q6_売上合計()

    Dim lastRow As Long, i As Long

    Set ws = ThisWorkbook.Worksheets("売上明細")

    Dim i As Long
    Dim cnt As Long

    Set ws = ThisWorkbook.Worksheets("売上明細")

    i = 2
    Do While ws.Cells(i, "A").Value <> ""

    Loop

    MsgBox "数量10個以上の明細は " & cnt & " 件です。" & vbLf & _

Sub Q8_給与ランク()

    Dim lastRow As Long, i As Long
    Dim salary As Long

            Case 300000 To 399999
                ws.Cells(i, "J").Value = "C"
            Case Else
                ws.Cells(i, "J").Value = "D"

    Next i

    ws.Columns("J").AutoFit

    MsgBox "支店別集計が終わりました。", vbInformation

Option Explicit

'==============================================================
' Lv3 実務でよく使う操作

'--- 共通: シートが存在するか調べる -------------------------
Public Function SheetExists(ByVal sheetName As String) As Boolean
    Dim ws As Worksheet
    For Each ws In ThisWorkbook.Worksheets
        If ws.Name = sheetName Then
            SheetExists = True
            Exit Function
        End If
    Next ws
End Function

'--- Q10: オートフィルタで抽出して別シートへ ----------------
Sub Q10_東京本社を抽出()
    Dim wsS As Worksheet, wsOut As Worksheet
    Dim lastRow As Long
    Const OUT_NAME As String = "東京本社_抽出"

    Set wsS = ThisWorkbook.Worksheets("売上明細")
    lastRow = wsS.Cells(wsS.Rows.Count, "A").End(xlUp).Row

    ' 出力先シートを作り直す
    Application.DisplayAlerts = False
    If SheetExists(OUT_NAME) Then ThisWorkbook.Worksheets(OUT_NAME).Delete
    Application.DisplayAlerts = True
    Set wsOut = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
    wsOut.Name = OUT_NAME

    ' 既存フィルタを解除してからかける
    If wsS.AutoFilterMode Then wsS.AutoFilterMode = False
    wsS.Range("A1:K" & lastRow).AutoFilter Field:=3, Criteria1:="東京本社"

    ' 可視セルだけコピー（該当0件だと落ちるので件数を確認）
    If wsS.Range("A1:A" & lastRow).SpecialCells(xlCellTypeVisible).Count > 1 Then
        wsS.Range("A1:K" & lastRow).SpecialCells(xlCellTypeVisible).Copy _
            Destination:=wsOut.Range("A1")
    Else
        MsgBox "該当データがありません。", vbExclamation
    End If

    wsS.AutoFilterMode = False
    wsOut.Columns("A:K").AutoFit
    wsOut.Activate
End Sub

'--- Q11: シートを安全に追加する ----------------------------
Sub Q11_月次シートを作る()
    Dim ws As Worksheet
    Dim sheetName As String

    sheetName = Format(Date, "yyyymm") & "_月次"

    If SheetExists(sheetName) Then
        If MsgBox(sheetName & " は既にあります。作り直しますか？", _
                  vbYesNo + vbQuestion) = vbNo Then Exit Sub
        Application.DisplayAlerts = False
        ThisWorkbook.Worksheets(sheetName).Delete
        Application.DisplayAlerts = True
    End If

    Set ws = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
    ws.Name = sheetName
    ws.Range("A1").Value = sheetName & " 集計表"
    ws.Range("A1").Font.Size = 14
    ws.Range("A1").Font.Bold = True
    ws.Tab.Color = RGB(31, 78, 121)
End Sub

'--- Q12: 並べ替え ------------------------------------------
Sub Q12_金額降順で並べ替え()
    Dim ws As Worksheet
    Dim lastRow As Long

    Set ws = ThisWorkbook.Worksheets("売上明細")
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    With ws.Sort
        .SortFields.Clear
        .SortFields.Add Key:=ws.Range("K2:K" & lastRow), Order:=xlDescending
        .SortFields.Add Key:=ws.Range("A2:A" & lastRow), Order:=xlAscending
        .SetRange ws.Range("A1:K" & lastRow)
        .Header = xlYes
        .Apply
    End With

    MsgBox "金額の降順（同額は日付の昇順）で並べ替えました。", vbInformation
End Sub

'--- Q13: マスタから単価を転記（VLOOKUP相当） ---------------
Sub Q13_単価を転記()
    Dim wsM As Worksheet, wsI As Worksheet
    Dim lastM As Long, lastI As Long, i As Long
    Dim v As Variant

    Set wsM = ThisWorkbook.Worksheets("商品マスタ")
    Set wsI = ThisWorkbook.Worksheets("在庫")
    lastM = wsM.Cells(wsM.Rows.Count, "A").End(xlUp).Row
    lastI = wsI.Cells(wsI.Rows.Count, "A").End(xlUp).Row

    For i = 2 To lastI
        ' 見つからないとエラーになるので、いったん Variant で受けて判定する
        v = Application.VLookup(wsI.Cells(i, "A").Value, _
                                wsM.Range("A2:D" & lastM), 4, False)
        If IsError(v) Then
            wsI.Cells(i, "F").Value = ""
            wsI.Cells(i, "F").Interior.Color = RGB(255, 235, 235)
        Else
            wsI.Cells(i, "F").Value = v
        End If
    Next i

    MsgBox "単価を転記しました。", vbInformation
End Sub

'--- Q14: 書式を整える定型処理 ------------------------------
Sub Q14_表を整形する()
    Dim ws As Worksheet
    Dim rng As Range
    Dim lastRow As Long, lastCol As Long

    Set ws = ActiveSheet
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column

    Set rng = ws.Range(ws.Cells(1, 1), ws.Cells(lastRow, lastCol))

    ' 罫線
    With rng.Borders
        .LineStyle = xlContinuous
        .Weight = xlThin
        .Color = RGB(191, 191, 191)
    End With

    ' 見出し

        .Interior.Color = RGB(31, 78, 121)
        .Font.Color = RGB(255, 255, 255)
        .Font.Bold = True
        .HorizontalAlignment = xlCenter
    End With

    ' 表示・印刷設定
    ws.Cells.EntireColumn.AutoFit
    ws.Range("A2").Select
    ActiveWindow.FreezePanes = False
    ActiveWindow.FreezePanes = True

    With ws.PageSetup
        .Orientation = xlLandscape
        .Zoom = False
        .FitToPagesWide = 1
        .FitToPagesTall = False
        .PrintTitleRows = "$1:$1"
    End With
End Sub

Option Explicit

'==============================================================

' ※ Scripting.Dictionary は Windows 版 Excel 専用です。

'--- Q15: Dictionary で担当者別集計（Windows） --------------
Sub Q15_担当者別集計()
    Dim ws As Worksheet, wsOut As Worksheet
    Dim dic As Object
    Dim data As Variant
    Dim i As Long, r As Long
    Dim key As String

    Set ws = ThisWorkbook.Worksheets("売上明細")
    Set wsOut = ThisWorkbook.Worksheets("集計結果")
    Set dic = CreateObject("Scripting.Dictionary")

    ' セルを1回だけ読んで配列に入れる（これが速さの肝）
    data = ws.Range("A1").CurrentRegion.Value

    For i = 2 To UBound(data, 1)
        key = data(i, 5)                       ' E列: 担当者名
        dic(key) = dic(key) + data(i, 11)      ' K列: 金額
    Next i

    wsOut.Range("G:H").ClearContents
    wsOut.Range("G1").Value = "担当者"
    wsOut.Range("H1").Value = "売上合計"
    wsOut.Range("G1:H1").Font.Bold = True

    r = 2
    Dim k As Variant
    For Each k In dic.Keys
        wsOut.Cells(r, "G").Value = k
        wsOut.Cells(r, "H").Value = dic(k)
        wsOut.Cells(r, "H").NumberFormat = "#,##0"
        r = r + 1
    Next k

    wsOut.Columns("G:H").AutoFit
    MsgBox dic.Count & " 人分を集計しました。", vbInformation
End Sub

'--- Q15': Mac でも動く Collection 版 -----------------------
Sub Q15_担当者別集計_Mac版()
    Dim ws As Worksheet, wsOut As Worksheet
    Dim names As Collection
    Dim data As Variant
    Dim sums() As Currency
    Dim i As Long, idx As Long, r As Long
    Dim key As String

    Set ws = ThisWorkbook.Worksheets("売上明細")
    Set wsOut = ThisWorkbook.Worksheets("集計結果")
    Set names = New Collection
    ReDim sums(1 To 1000)

    data = ws.Range("A1").CurrentRegion.Value

    For i = 2 To UBound(data, 1)
        key = CStr(data(i, 5))
        idx = IndexOfKey(names, key)
        If idx = 0 Then
            names.Add key
            idx = names.Count
        End If
        sums(idx) = sums(idx) + data(i, 11)
    Next i

    wsOut.Range("G:H").ClearContents
    wsOut.Range("G1").Value = "担当者"
    wsOut.Range("H1").Value = "売上合計"
    wsOut.Range("G1:H1").Font.Bold = True

    For r = 1 To names.Count
        wsOut.Cells(r + 1, "G").Value = names(r)
        wsOut.Cells(r + 1, "H").Value = sums(r)
        wsOut.Cells(r + 1, "H").NumberFormat = "#,##0"
    Next r

    wsOut.Columns("G:H").AutoFit
    MsgBox names.Count & " 人分を集計しました。", vbInformation
End Sub

Private Function IndexOfKey(ByVal col As Collection, ByVal key As String) As Long
    Dim i As Long
    For i = 1 To col.Count
        If col(i) = key Then
            IndexOfKey = i
            Exit Function
        End If
    Next i
    IndexOfKey = 0
End Function

'--- Q16: 配列で高速化＋速度比較 ----------------------------
Sub Q16_速度比較()
    Dim ws As Worksheet
    Dim lastRow As Long, i As Long
    Dim t1 As Double, t2 As Double
    Dim data As Variant
    Dim total As Currency

    Set ws = ThisWorkbook.Worksheets("売上明細")

    ' 方法1: 1セルずつ読む（遅い）

    total = 0
    For i = 2 To lastRow
        total = total + ws.Cells(i, "K").Value
    Next i

    t2 = Timer
    data = ws.Range("K2:K" & lastRow).Value

    For i = 1 To UBound(data, 1)

    Next i
    t2 = Timer - t2

    MsgBox "セル直接 : " & Format(t1, "0.000") & " 秒" & vbLf & _
           "配列経由 : " & Format(t2, "0.000") & " 秒" & vbLf & vbLf & _
           "合計金額 : " & Format(total, "#,##0") & " 円"

'--- Q18: ユーザー定義関数 ----------------------------------
' セルに =税込(A1) のように書いて使えます
Public Function 税込(ByVal amount As Double, Optional ByVal rate As Double = 0.1) As Double
    税込 = Int(amount * (1 + rate))
End Function

' 全角英数字・カナを半角に、前後の空白を除去
Public Function 正規化(ByVal s As Variant) As String
    If IsError(s) Or IsEmpty(s) Then
        正規化 = ""
        Exit Function
    End If
    Dim t As String
    t = CStr(s)
    t = StrConv(t, vbNarrow)          ' 全角 → 半角
    t = Replace(t, vbTab, " ")
    t = Replace(t, "　", " ")          ' 全角スペース → 半角
    Do While InStr(t, "  ") > 0
        t = Replace(t, "  ", " ")      ' 連続スペースを1つに
    Loop
    正規化 = Trim(t)
End Function

' 年度（4月始まり）を返す
Public Function 年度(ByVal d As Date) As Long
    If Month(d) >= 4 Then
        年度 = Year(d)
    Else
        年度 = Year(d) - 1
    End If
End Function

'--- Q19: 汚いデータのクレンジング --------------------------
Sub Q19_データを整える()
    Dim wsSrc As Worksheet, wsOut As Worksheet
    Dim lastRow As Long, i As Long, r As Long
    Dim seen As Collection
    Dim code As String
    Const OUT_NAME As String = "整形済みデータ"

    Set wsSrc = ThisWorkbook.Worksheets("要整形データ")
    lastRow = wsSrc.Cells(wsSrc.Rows.Count, "A").End(xlUp).Row

    Application.DisplayAlerts = False
    On Error Resume Next
    ThisWorkbook.Worksheets(OUT_NAME).Delete
    On Error GoTo 0
    Application.DisplayAlerts = True

    Set wsOut = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
    wsOut.Name = OUT_NAME
    wsOut.Range("A1:G1").Value = Array("顧客番号", "会社名", "電話番号", "郵便番号", "住所", "契約金額", "契約日")
    wsOut.Range("A1:G1").Font.Bold = True

    Set seen = New Collection
    r = 2

    For i = 2 To lastRow
        code = UCase(正規化(wsSrc.Cells(i, "A").Value))

        If code <> "" Then
            ' 重複チェック（Collection のキー重複エラーを利用する定番手法）
            On Error Resume Next
            seen.Add code, code
            If Err.Number = 0 Then
                On Error GoTo 0
                wsOut.Cells(r, "A").Value = code
                wsOut.Cells(r, "B").Value = 正規化(wsSrc.Cells(i, "B").Value)
                wsOut.Cells(r, "C").Value = Replace(Replace(正規化(wsSrc.Cells(i, "C").Value), "‐", "-"), "−", "-")
                wsOut.Cells(r, "D").Value = 郵便番号整形(wsSrc.Cells(i, "D").Value)
                wsOut.Cells(r, "E").Value = 正規化(wsSrc.Cells(i, "E").Value)
                wsOut.Cells(r, "F").Value = 金額に変換(wsSrc.Cells(i, "F").Value)
                wsOut.Cells(r, "G").Value = 日付に変換(wsSrc.Cells(i, "G").Value)
                wsOut.Cells(r, "F").NumberFormat = "#,##0"
                wsOut.Cells(r, "G").NumberFormat = "yyyy/mm/dd"
                r = r + 1
            End If
            Err.Clear
            On Error GoTo 0
        End If
    Next i

    wsOut.Columns("A:G").AutoFit
    MsgBox (r - 2) & " 件に整形しました（空行・重複を除去）。", vbInformation
End Sub

Private Function 郵便番号整形(ByVal v As Variant) As String
    Dim s As String
    s = Replace(正規化(v), "-", "")
    If Len(s) = 7 Then
        郵便番号整形 = Left(s, 3) & "-" & Right(s, 4)
    Else
        郵便番号整形 = s
    End If
End Function

Private Function 金額に変換(ByVal v As Variant) As Variant
    Dim s As String
    s = 正規化(v)
    s = Replace(s, ",", "")
    s = Replace(s, "，", "")
    s = Replace(s, "円", "")
    s = Replace(s, " ", "")
    If IsNumeric(s) Then
        金額に変換 = CDbl(s)
    Else
        金額に変換 = s
    End If
End Function

Private Function 日付に変換(ByVal v As Variant) As Variant
    Dim s As String
    s = 正規化(v)
    s = Replace(s, "年", "/")
    s = Replace(s, "月", "/")
    s = Replace(s, "日", "")
    s = Replace(s, ".", "/")

    ' 和暦（R7 → 2025）
    If UCase(Left(s, 1)) = "R" Then
        Dim parts As Variant
        parts = Split(Mid(s, 2), "/")
        If UBound(parts) >= 2 Then
            s = (2018 + CLng(parts(0))) & "/" & parts(1) & "/" & parts(2)
        End If
    End If

    If IsDate(s) Then
        日付に変換 = CDate(s)
    Else
        日付に変換 = s
    End If
End Function

Option Explicit

'==============================================================
' Lv5 総合課題
'   データ フォルダの CSV を全部取り込む
'     → 支店 × カテゴリ のクロス集計
'     → 請求書テンプレートへ書き出し → PDF 出力
'
' ※ Mac の場合、Dir でフォルダを走査する前に一度
'    GrantAccessToMultipleFiles でアクセス許可が必要になることがあります。
'==============================================================

Private Const CSV_FOLDER As String = "データ"

'--- Q20-1: フォルダ内の CSV をすべて1シートに取り込む ------
Sub Q20_1_CSVを全部取り込む()
    Dim folderPath As String, fileName As String
    Dim wbCsv As Workbook
    Dim wsOut As Worksheet
    Dim lastRowSrc As Long, nextRow As Long
    Dim fileCount As Long
    Const OUT_NAME As String = "取込データ"

    folderPath = ThisWorkbook.Path & Application.PathSeparator & CSV_FOLDER & Application.PathSeparator

    Application.ScreenUpdating = False
    On Error GoTo ErrHandler

    ' 出力シートを作り直す
    Application.DisplayAlerts = False
    On Error Resume Next
    ThisWorkbook.Worksheets(OUT_NAME).Delete
    On Error GoTo ErrHandler
    Application.DisplayAlerts = True
    Set wsOut = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
    wsOut.Name = OUT_NAME

    nextRow = 1
    fileName = Dir(folderPath & "*.csv")

    If fileName = "" Then
        MsgBox "CSV が見つかりません。" & vbLf & folderPath, vbExclamation
        GoTo CleanUp
    End If

    Do While fileName <> ""
        Set wbCsv = Workbooks.Open(fileName:=folderPath & fileName, ReadOnly:=True)
        With wbCsv.Worksheets(1)
            lastRowSrc = .Cells(.Rows.Count, "A").End(xlUp).Row
            If nextRow = 1 Then
                ' 1ファイル目は見出しごとコピー
                .Range("A1:K" & lastRowSrc).Copy wsOut.Cells(1, 1)
                nextRow = lastRowSrc + 1
            Else
                ' 2ファイル目以降は見出しを飛ばす
                .Range("A2:K" & lastRowSrc).Copy wsOut.Cells(nextRow, 1)
                nextRow = nextRow + lastRowSrc - 1
            End If
        End With
        wbCsv.Close SaveChanges:=False
        fileCount = fileCount + 1
        fileName = Dir                      ' ← 次のファイルへ
    Loop

    wsOut.Columns("A:K").AutoFit
    MsgBox fileCount & " 個の CSV を取り込みました（" & (nextRow - 2) & " 行）。", vbInformation

CleanUp:
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    Exit Sub

ErrHandler:
    MsgBox "取込中にエラー: " & Err.Description, vbCritical
    If Not wbCsv Is Nothing Then wbCsv.Close SaveChanges:=False
    Resume CleanUp
End Sub

'--- Q20-2: 支店 × カテゴリ のクロス集計 --------------------
Sub Q20_2_クロス集計()
    Dim wsIn As Worksheet, wsOut As Worksheet
    Dim data As Variant
    Dim branches As Collection, cats As Collection
    Dim grid() As Currency
    Dim i As Long, bi As Long, ci As Long
    Const IN_NAME As String = "取込データ"
    Const OUT_NAME As String = "クロス集計"

    On Error Resume Next
    Set wsIn = ThisWorkbook.Worksheets(IN_NAME)
    On Error GoTo 0
    If wsIn Is Nothing Then
        MsgBox "先に Q20_1_CSVを全部取り込む を実行してください。", vbExclamation
        Exit Sub
    End If

    data = wsIn.Range("A1").CurrentRegion.Value

    Set branches = New Collection
    Set cats = New Collection
    For i = 2 To UBound(data, 1)
        AddUnique branches, CStr(data(i, 3))    ' C列: 支店
        AddUnique cats, CStr(data(i, 8))        ' H列: カテゴリ
    Next i

    ReDim grid(1 To branches.Count, 1 To cats.Count)
    For i = 2 To UBound(data, 1)
        bi = FindIndex(branches, CStr(data(i, 3)))
        ci = FindIndex(cats, CStr(data(i, 8)))
        grid(bi, ci) = grid(bi, ci) + data(i, 11)
    Next i

    Application.DisplayAlerts = False
    On Error Resume Next
    ThisWorkbook.Worksheets(OUT_NAME).Delete
    On Error GoTo 0
    Application.DisplayAlerts = True
    Set wsOut = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
    wsOut.Name = OUT_NAME

    wsOut.Range("A1").Value = "支店＼カテゴリ"
    For ci = 1 To cats.Count
        wsOut.Cells(1, ci + 1).Value = cats(ci)
    Next ci
    wsOut.Cells(1, cats.Count + 2).Value = "合計"

    For bi = 1 To branches.Count
        wsOut.Cells(bi + 1, 1).Value = branches(bi)
        For ci = 1 To cats.Count
            wsOut.Cells(bi + 1, ci + 1).Value = grid(bi, ci)
        Next ci
        wsOut.Cells(bi + 1, cats.Count + 2).Formula = _
            "=SUM(B" & (bi + 1) & ":" & Chr(65 + cats.Count) & (bi + 1) & ")"
    Next bi

    With wsOut.Range(wsOut.Cells(1, 1), wsOut.Cells(branches.Count + 1, cats.Count + 2))
        .Borders.LineStyle = xlContinuous
        .Borders.Color = RGB(191, 191, 191)
    End With
    wsOut.Range(wsOut.Cells(1, 1), wsOut.Cells(1, cats.Count + 2)).Interior.Color = RGB(31, 78, 121)
    wsOut.Range(wsOut.Cells(1, 1), wsOut.Cells(1, cats.Count + 2)).Font.Color = RGB(255, 255, 255)
    wsOut.Range(wsOut.Cells(1, 1), wsOut.Cells(1, cats.Count + 2)).Font.Bold = True
    wsOut.Range(wsOut.Cells(2, 2), wsOut.Cells(branches.Count + 1, cats.Count + 2)).NumberFormat = "#,##0"
    wsOut.Columns.AutoFit
    wsOut.Activate
End Sub

Private Sub AddUnique(ByVal col As Collection, ByVal key As String)
    On Error Resume Next
    col.Add key, key      ' キー重複ならエラー → 追加されない
    Err.Clear
End Sub

Private Function FindIndex(ByVal col As Collection, ByVal key As String) As Long
    Dim i As Long
    For i = 1 To col.Count
        If col(i) = key Then
            FindIndex = i
            Exit Function
        End If
    Next i
End Function

'--- Q20-3: 請求書テンプレートへ書き出して PDF 出力 ---------
Sub Q20_3_請求書を発行()
    Dim wsT As Worksheet, wsS As Worksheet
    Dim data As Variant
    Dim targetSlip As String
    Dim i As Long, r As Long
    Dim total As Currency
    Dim pdfPath As String

    Set wsT = ThisWorkbook.Worksheets("請求書テンプレート")
    Set wsS = ThisWorkbook.Worksheets("売上明細")

    targetSlip = InputBox("請求書にする伝票番号を入力してください（例: S10001）", "伝票番号", "S10001")
    If targetSlip = "" Then Exit Sub

    ' 明細欄をクリア
    wsT.Range("B10:F24").ClearContents

    data = wsS.Range("A1").CurrentRegion.Value
    r = 10
    total = 0

    For i = 2 To UBound(data, 1)
        If CStr(data(i, 2)) = targetSlip Then
            If r > 24 Then
                MsgBox "明細が15行を超えました。", vbExclamation
                Exit For
            End If
            wsT.Cells(r, "B").Value = r - 9            ' No
            wsT.Cells(r, "C").Value = data(i, 7)       ' 商品名
            wsT.Cells(r, "D").Value = data(i, 10)      ' 数量
            wsT.Cells(r, "E").Value = data(i, 9)       ' 単価
            wsT.Cells(r, "F").Value = data(i, 11)      ' 金額
            wsT.Cells(r, "E").NumberFormat = "#,##0"
            wsT.Cells(r, "F").NumberFormat = "#,##0"
            total = total + data(i, 11)

            wsT.Range("C4").Value = data(i, 5) & " 様"  ' 担当者名を宛先に流用
            wsT.Range("C5").Value = data(i, 3) & " 分 売上"
            wsT.Range("F4").Value = data(i, 1)
            wsT.Range("F4").NumberFormat = "yyyy/mm/dd"
            wsT.Range("F5").Value = targetSlip
            r = r + 1
        End If
    Next i

    If total = 0 Then
        MsgBox "伝票番号 " & targetSlip & " が見つかりませんでした。", vbExclamation
        Exit Sub
    End If

    wsT.Range("C7").Value = Int(total * 1.1)

    ' PDF 出力
    pdfPath = ThisWorkbook.Path & Application.PathSeparator & "請求書_" & targetSlip & ".pdf"
    wsT.PageSetup.Orientation = xlPortrait
    wsT.ExportAsFixedFormat Type:=xlTypePDF, fileName:=pdfPath, Quality:=xlQualityStandard

    MsgBox "請求書を出力しました。" & vbLf & pdfPath, vbInformation
End Sub
